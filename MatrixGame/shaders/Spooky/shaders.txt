# The world's depth as the moon sees it, for the rays (drawn only while they are on: the world is drawn twice).

# The world drawn by world.hlsl: the surfaces' normals, what each thing is (only while GEOMETRY is on).
Target.WorldN=1|rgba8
Target.WorldM=1|rgba8
World=world.hlsl|WorldN WorldM|GEOMETRY

# The effects drawn by its own shader in the game's place.
Effects=effects.hlsl|SOFTFX

Target.Half=0.5|rgba16f
Target.HalfDir=0.5|rgba16f
Target.Glow=0.25|rgba16f
Target.GlowH=0.25|rgba16f

Target.Wide=0.0625|rgba16f
Target.Lit=1|rgba16f
Target.Rays=0.5|rgba8
Target.RaysB=0.5|rgba8

Pass=halfdir.hlsl|HalfDir
